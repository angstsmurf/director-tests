-- A key ends a wait-for-click tempo.
-- Frame 2 waits for a mouse click. Frame 1 posts a key and goes to frame 2;
-- frame 3's enterFrame measures how long the wait took.
-- Stage 1: the key is due at once (queued before the go).
-- Stage 2: the key is due 1.5 s later (posted during the wait).
-- Real D5 Win ends the wait on keyDown in its event dispatcher and still
-- sends the key to Lingo. (No clicks: the projector ignores posted
-- WM_LBUTTONDOWN messages.)
global gStage, gEv, gT0, gDur, gAfter, gTempo

on startMovie
  set gStage = 1
  set gAfter = 0
  set gEv = []
  UTBeginTest("a key ends a wait for a click")
end

on keyDown
  append gEv, "key"
end

on mouseDown
  append gEv, "down"
end

on mouseUp
  append gEv, "up"
end

on enterFrame
  if the frame = 2 then set gTempo = the frameTempo
  if the frame = 3 and gAfter = 0 then
    set gDur = the ticks - gT0
    set gAfter = the ticks
  end if
end

on checkStage
  set d = gDur
  set e = string(gEv)
  UTLog("stage " & gStage & " frame 2 tempo " & gTempo & ", wait took " & d & " ticks, events " & e)
  if gStage = 1 then UTAssertTrue(d < 30, "a key queued before the go ends the wait at once (" & d & ")")
  if gStage = 2 then UTAssertTrue(d > 75 and d < 110, "a key 1.5 s in ends the wait then (" & d & ")")
  UTAssertEqual("[" & QUOTE & "key" & QUOTE & "]", e, "stage " & gStage & ": the key reaches Lingo")
end

on exitFrame
  set the timeoutLapsed to 0
  if the frame = 1 then
    if gStage > 2 then
      UTEndTest(4)
      exit
    end if
    set gEv = []
    set gAfter = 0
    if gStage = 1 then UTPostKey(65, 0)
    if gStage = 2 then UTPostKey(65, 1500)
    set gT0 = the ticks
    go to frame 2
  else if the frame = 3 then
    if the ticks < gAfter + 30 then
      go to the frame
    else
      checkStage()
      set gStage = gStage + 1
      go to frame 1
    end if
  end if
end
