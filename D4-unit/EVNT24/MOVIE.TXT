global evCount, phase

on startMovie
  UTBeginTest("a slow go from idle exits the frame before the next idle")
  set evCount to 1
  set phase to 0
  puppetTempo 60
  UTAssertEqual(1, evCount, "startMovie")
  set evCount to evCount + 1
end

on idle
  if the frame = 3 then
    if phase = 1 then
      set phase to 2
      UTAssertEqual(4, evCount, "idle on frame 3")
      set evCount to evCount + 1
      go(4)
      UTAssertEqual(6, evCount, "idle after go 4")
      set evCount to evCount + 1
    end if
  else
    if (the frame = 4) and (phase = 2) then
      set phase to 3
      UTAssertTrue(0, "idle on frame 4 before its exitFrame")
    end if
  end if
end
