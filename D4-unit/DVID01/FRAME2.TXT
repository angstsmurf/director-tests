-- Sprite 1 is cast 7, a linked QuickTime movie (DVA.MOV, 160x120, 1 s = 60 ticks)
global nx
on exitFrame
  set nx to nx + 1
  if nx = 30 then
    UTAssertEqual(0, the movieRate of sprite 1, "video stopped at its end")
    UTAssertEqual(60, the movieTime of sprite 1, "movieTime at the end")
    set the movieRate of sprite 1 to 0
    set the movieTime of sprite 1 to 0
  end if
  if nx = 34 then
    UTAssertEqual(0, the movieTime of sprite 1, "movieTime after rewinding to 0")
    set the movieRate of sprite 1 to 1
  end if
  if nx = 38 then
    UTAssertTrue(the movieTime of sprite 1 > 0, "playing again after the rewind")
    UTAssertTrue(the movieTime of sprite 1 < 60, "not at the end after the rewind")
  end if
  if nx = 70 then
    UTAssertEqual(60, the movieTime of sprite 1, "replayed to the end")
    UTEndTest(6)
  end if
  if nx < 70 then go to the frame
end
