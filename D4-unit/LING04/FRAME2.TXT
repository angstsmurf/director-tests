on exitFrame
  set e to numToChar(233)
  UTAssertEqual(1, offset("", "abc"), "empty target")
  UTAssertEqual(1, offset("", ""), "empty target and source")
  UTAssertEqual(0, offset("a", ""), "empty source")
  UTAssertEqual(1, offset("a", "abc"), "match at position 1")
  UTAssertEqual(3, offset("c", "abc"), "match at the end")
  UTAssertEqual(1, offset("abc", "abc"), "whole string")
  UTAssertEqual(0, offset("d", "abc"), "no match")
  UTAssertEqual(0, offset("abcd", "abc"), "target longer than source")
  UTAssertEqual(0, offset("abc", "xab"), "partial match at the end")
  UTAssertEqual(7, offset("File", "Fi le File"), "restart after a partial match")
  UTAssertEqual(2, offset("aab", "aaab"), "repeated characters")
  UTAssertEqual(3, offset("ss", "mississippi"), "first of two matches")
  UTAssertEqual(2, offset(" ", "a b"), "space")
  UTAssertEqual(3, offset("FILE", "a file"), "ignores case")
  UTAssertEqual(2, offset("x", e & "x"), "counts characters, not bytes")
  UTAssertEqual(2, offset("e", "x" & e), "ignores accents")
  UTAssertEqual(2, offset(numToChar(201), "x" & e), "ignores accents and case")
  UTEndTest(17)
end
